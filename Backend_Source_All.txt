using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
using Handicraft.Domain.Entities;
using BCrypt.Net;
using Handicraft.Application.Contracts.Persistence;
using Handicraft.Api.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Authorization;
using Microsoft.IdentityModel.Tokens;

namespace Handicraft.Api.Controllers;

[ApiController]
[Route("api/[controller]")]
public class AuthController : ControllerBase
{
    private readonly IUserRepository _userRepository;
    private readonly IRefreshTokenRepository _refreshTokenRepository;
    private readonly IConfiguration _configuration;

    public AuthController(IUserRepository userRepository, IRefreshTokenRepository refreshTokenRepository, IConfiguration configuration)
    {
        _userRepository = userRepository;
        _refreshTokenRepository = refreshTokenRepository;
        _configuration = configuration;
    }

    [HttpPost("revoke")]
    public async Task<IActionResult> Revoke([FromBody] RefreshRequestDto dto)
    {
        if (dto == null || string.IsNullOrWhiteSpace(dto.Token)) return BadRequest();
        if (_refreshTokenRepository == null) return StatusCode(500, new { message = "Refresh token repository not available" });

        var stored = await _refreshTokenRepository.GetByTokenAsync(dto.Token);
        if (stored == null) return NotFound();

        await _refreshTokenRepository.RevokeAsync(stored);
        return NoContent();
    }

    [Authorize]
    [HttpPost("logout")]
    public async Task<IActionResult> Logout()
    {
        var userIdClaim = User.FindFirst(System.IdentityModel.Tokens.Jwt.JwtRegisteredClaimNames.Sub)?.Value;
        if (!Guid.TryParse(userIdClaim, out var userId)) return Unauthorized();

        if (_refreshTokenRepository == null) return StatusCode(500, new { message = "Refresh token repository not available" });

        await _refreshTokenRepository.RevokeAllForUserAsync(userId);
        return NoContent();
    }

    [HttpPost("register")]
    public async Task<IActionResult> Register([FromBody] RegisterDto dto)
    {
        if (dto == null) return BadRequest();
        // basic validation
        if (string.IsNullOrWhiteSpace(dto.Password) || dto.Password.Length < 8)
            return BadRequest(new { message = "Password must be at least 8 characters." });

        var unique = await _userRepository.IsPhoneUniqueAsync(dto.Phone);
        if (!unique) return BadRequest(new { message = "Phone already registered" });

        var user = new Domain.Entities.User
        {
            Id = Guid.NewGuid(),
            Phone = dto.Phone,
            Name = dto.Name,
            Role = dto.Role ?? "User",
            IsActive = true,
            CreatedAt = DateTime.UtcNow,
            PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.Password)
        };

        await _userRepository.AddAsync(user);
        // if repository does not save changes, ensure caller saves; our GenericRepository uses context and SaveChanges may be required.
        // Attempt to save via a minimal approach if repository exposes no SaveChanges: rely on unit of work not present — use context via cast if needed.

        var token = GenerateToken(user);
        // create refresh token
        var refresh = CreateRefreshToken(user.Id);
        if (_refreshTokenRepository != null)
        {
            await _refreshTokenRepository.AddAsync(refresh);
        }

        return Ok(new AuthResponseDto { Token = token, ExpiresInMinutes = _configuration.GetValue<int>("Jwt:ExpiresMinutes"), RefreshToken = refresh.Token });
    }

    [Authorize]
    [HttpPost("change-password")]
    public async Task<IActionResult> ChangePassword([FromBody] ChangePasswordDto dto)
    {
        if (dto == null) return BadRequest();

        var userIdClaim = User.FindFirst(System.IdentityModel.Tokens.Jwt.JwtRegisteredClaimNames.Sub)?.Value;
        if (!Guid.TryParse(userIdClaim, out var userId)) return Unauthorized();

        var user = await _userRepository.GetByIdAsync(userId);
        if (user == null) return NotFound();

        var verified = BCrypt.Net.BCrypt.Verify(dto.OldPassword, user.PasswordHash);
        if (!verified) return BadRequest(new { message = "Old password is incorrect" });

        if (string.IsNullOrWhiteSpace(dto.NewPassword) || dto.NewPassword.Length < 8)
            return BadRequest(new { message = "New password must be at least 8 characters." });

        user.PasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.NewPassword);
        await _userRepository.UpdateAsync(user);

        // revoke existing refresh tokens for safety
        if (_refreshTokenRepository != null)
        {
            await _refreshTokenRepository.RevokeAllForUserAsync(user.Id);
        }

        return NoContent();
    }

    [HttpPost("refresh")]
    public async Task<IActionResult> Refresh([FromBody] RefreshRequestDto dto)
    {
        // refresh using persistent refresh tokens
        if (dto == null || string.IsNullOrWhiteSpace(dto.Token)) return BadRequest();
        if (_refreshTokenRepository == null) return StatusCode(500, new { message = "Refresh token repository not available" });

        var stored = await _refreshTokenRepository.GetByTokenAsync(dto.Token);
        if (stored == null) return Unauthorized();
        if (stored.ExpiresAt < DateTime.UtcNow) return Unauthorized(new { message = "Refresh token expired" });

        var user = await _userRepository.GetByIdAsync(stored.UserId);
        if (user == null) return Unauthorized();

        // rotate refresh token
        stored.IsRevoked = true;
        var newRefresh = CreateRefreshToken(user.Id);
        stored.ReplacedByToken = newRefresh.Token;
        await _refreshTokenRepository.UpdateAsync(stored);
        await _refreshTokenRepository.AddAsync(newRefresh);

        var newAccess = GenerateToken(user);
        return Ok(new AuthResponseDto { Token = newAccess, ExpiresInMinutes = _configuration.GetValue<int>("Jwt:ExpiresMinutes"), RefreshToken = newRefresh.Token });
    }

    [HttpPost("login")]
    public async Task<IActionResult> Login([FromBody] LoginDto dto)
    {
        if (dto == null) return BadRequest();

        var user = await _userRepository.GetByPhoneAsync(dto.Phone);
        if (user == null) return Unauthorized(new { message = "Invalid credentials" });

        var verified = BCrypt.Net.BCrypt.Verify(dto.Password, user.PasswordHash);
        if (!verified) return Unauthorized(new { message = "Invalid credentials" });

        var token = GenerateToken(user);
        // create refresh token
        var refresh = CreateRefreshToken(user.Id);
        if (_refreshTokenRepository != null)
        {
            await _refreshTokenRepository.AddAsync(refresh);
        }

        return Ok(new AuthResponseDto { Token = token, ExpiresInMinutes = _configuration.GetValue<int>("Jwt:ExpiresMinutes"), RefreshToken = refresh.Token });
    }

    private string GenerateToken(Domain.Entities.User user)
    {
        var jwtSection = _configuration.GetSection("Jwt");
        var key = jwtSection.GetValue<string>("Key");
        var issuer = jwtSection.GetValue<string>("Issuer");
        var audience = jwtSection.GetValue<string>("Audience");
        var expires = jwtSection.GetValue<int>("ExpiresMinutes");

        var claims = new List<Claim>
        {
            new Claim(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
            new Claim(JwtRegisteredClaimNames.UniqueName, user.Phone),
            new Claim(ClaimTypes.Role, user.Role ?? string.Empty)
        };

        var keyBytes = Encoding.UTF8.GetBytes(key);
        var tokenDescriptor = new SecurityTokenDescriptor
        {
            Subject = new ClaimsIdentity(claims),
            Expires = DateTime.UtcNow.AddMinutes(expires > 0 ? expires : 60),
            Issuer = issuer,
            Audience = audience,
            SigningCredentials = new SigningCredentials(new SymmetricSecurityKey(keyBytes), SecurityAlgorithms.HmacSha256Signature)
        };

        var tokenHandler = new JwtSecurityTokenHandler();
        var token = tokenHandler.CreateToken(tokenDescriptor);
        return tokenHandler.WriteToken(token);
    }

    private RefreshToken CreateRefreshToken(Guid userId)
    {
        var random = new byte[64];
        using var rng = System.Security.Cryptography.RandomNumberGenerator.Create();
        rng.GetBytes(random);
        var token = Convert.ToBase64String(random);

        var expiresDays = _configuration.GetValue<int>("Jwt:RefreshExpiresDays", 7);

        return new RefreshToken
        {
            Id = Guid.NewGuid(),
            UserId = userId,
            Token = token,
            CreatedAt = DateTime.UtcNow,
            ExpiresAt = DateTime.UtcNow.AddDays(expiresDays),
            IsRevoked = false
        };
    }
}

using Handicraft.Application.Contracts.Services;
using Handicraft.Application.Services.Dtos;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace Handicraft.Api.Controllers;

[ApiController]
[Route("api/admin/lessons/{lessonId:guid}/video")]
public class CourseEpisodesController : ControllerBase
{
    private readonly ICourseEpisodeService _episodeService;

    public CourseEpisodesController(
        ICourseEpisodeService episodeService)
    {
        _episodeService = episodeService;
    }


    [HttpPost]
    [Authorize(Roles = "Admin,Instructor")]
    [RequestSizeLimit(2_000_000_000)]
    public async Task<IActionResult> UploadEpisode(
        Guid lessonId,
        [FromForm] CourseEpisodeUploadDto dto)
    {
        var result =
            await _episodeService.CreateAndUploadAsync(
                lessonId,
                dto);

        return Ok(result);
    }


    [HttpGet("{episodeId:guid}/status")]
    public async Task<IActionResult> GetStatus(
        Guid episodeId)
    {
        var status =
            await _episodeService.GetProcessingStatusAsync(
                episodeId);

        if (status == null)
            return NotFound();

        return Ok(status);
    }


    [HttpGet("{episodeId:guid}/stream/{**path}")]
    [Authorize]
    public async Task<IActionResult> StreamHls(
        Guid lessonId,
        Guid episodeId,
        string path)
    {
        var hasAccess =
            await _episodeService.UserHasAccessAsync(
                User,
                lessonId,
                episodeId);


        if (!hasAccess)
            return Forbid();


        var fullPath =
            await _episodeService.ResolveHlsFilePathAsync(
                lessonId,
                episodeId,
                path);


        if (string.IsNullOrEmpty(fullPath) ||
            !System.IO.File.Exists(fullPath))
        {
            return NotFound();
        }


        var contentType =
            path.EndsWith(".m3u8")
                ? "application/vnd.apple.mpegurl"
                : "video/mp2t";


        return PhysicalFile(
            fullPath,
            contentType);
    }
}

using Handicraft.Application.Contracts.Persistence; // namespace ????
using Handicraft.Application.Contracts.Services;
using Handicraft.Application.Services.Dtos;
using Handicraft.Domain.Entities;
using System.Security.Claims;

namespace Handicraft.Application.Services;

public class CourseEpisodeService : ICourseEpisodeService
{
    private readonly ICourseEpisodeRepository _episodeRepository;
    private readonly IVideoStorageService _videoStorage;

    public CourseEpisodeService(
        ICourseEpisodeRepository episodeRepository,
        IVideoStorageService videoStorage)
    {
        _episodeRepository = episodeRepository;
        _videoStorage = videoStorage;
    }

    public async Task<CourseEpisodeDto> CreateAndUploadAsync(
    Guid lessonId,
    CourseEpisodeUploadDto dto)
    {
        var episode = new CourseEpisode
        {
            Id = Guid.NewGuid(),

            LessonId = lessonId,

            Title = dto.Title,

            Description = dto.Description,

            VideoUrl = string.Empty,

            Order = dto.Order,

            DurationSeconds = 0,

            IsFree = dto.IsFree,

            Status = VideoProcessingStatus.Pending,

            CreatedAt = DateTime.UtcNow
        };

        await using var stream = dto.VideoFile.OpenReadStream();
        await _videoStorage.SaveRawVideoAsync(  lessonId,
                                                episode.Id,
                                                stream,
                                                dto.VideoFile.FileName);

        episode.RawVideoFileName = dto.VideoFile.FileName;



        var created = await _episodeRepository.AddAsync(episode);

        return MapToDto(created);
    }

    public async Task<CourseEpisodeDto?> GetProcessingStatusAsync(Guid episodeId)
    {
        var episode = await _episodeRepository.GetByIdAsync(episodeId);
        return episode is null ? null : MapToDto(episode);
    }

    public async Task<bool> UserHasAccessAsync(
     ClaimsPrincipal user,
     Guid lessonId,
     Guid episodeId)
    {
        var episode = await _episodeRepository.GetByIdAsync(episodeId);

        if (episode is null)
            return false;

        if (episode.IsFree)
            return true;

        // ????? Enrollment ?? ?? Lesson -> Section -> Course ?? ???????
        return false;
    }

    public async Task<string?> ResolveHlsFilePathAsync(
      Guid lessonId,
      Guid episodeId,
      string relativePath)
    {
        var episode = await _episodeRepository.GetByIdAsync(episodeId);

        if (episode is null ||
            episode.Status != VideoProcessingStatus.Ready)
            return null;


        var hlsFolder = Path.GetDirectoryName(
            _videoStorage.GetHlsPlaylistPath(
                lessonId,
                episodeId)) ?? string.Empty;


        return Path.Combine(
            hlsFolder,
            relativePath);
    }

    public async Task<IReadOnlyList<CourseEpisodeDto>> GetEpisodesByLessonIdAsync(  Guid lessonId)
    {
        var episodes = await _episodeRepository
            .GetEpisodesByLessonIdAsync(lessonId);

        return episodes
            .Select(MapToDto)
            .ToList();
    }

    private static CourseEpisodeDto MapToDto(CourseEpisode e) => new(
        e.Id,
        e.Title,
        e.Description,
        e.VideoUrl,
        e.Order,
        e.DurationSeconds,
        e.IsFree
    );
}

using Handicraft.Application.Contracts.Persistence;
using Handicraft.Application.Contracts.Services;
using Handicraft.Application.Services.Dtos;
using Handicraft.Domain.Entities;

namespace Handicraft.Application.Services;

public class CourseService : ICourseService
{
    private readonly ICourseRepository _courseRepository;

    public CourseService(ICourseRepository courseRepository)
    {
        _courseRepository = courseRepository;
    }


    public async Task<CourseDto> CreateAsync(CourseCreateDto dto)
    {
        var course = new Course
        {
            Id = Guid.NewGuid(),
            Title = dto.Title,
            Description = dto.Description,
            CoverImageUrl = dto.CoverImageUrl,
            Price = dto.Price,
            IsPublished = dto.IsPublished,
            DurationMinutes = dto.DurationMinutes,
            CreatedAt = DateTime.UtcNow
        };

        var created = await _courseRepository.AddAsync(course);

        return MapToDto(created);
    }


    public async Task<IEnumerable<CourseDto>> GetAllAsync()
    {
        var courses = await _courseRepository.GetAllWithContentAsync();

        return courses.Select(MapToDto).ToList();
    }


    public async Task<IEnumerable<CourseDto>> GetPublishedAsync()
    {
        var courses = await _courseRepository.GetPublishedCoursesAsync();

        return courses.Select(MapToDto).ToList();
    }


    public async Task<CourseDto?> GetByIdAsync(Guid id)
    {
        var course = await _courseRepository.GetByIdWithContentAsync(id);

        if (course == null)
            return null;

        return MapToDto(course);
    }


    private static CourseDto MapToDto(Course c)
    {
        var sections = c.Sections?
            .OrderBy(x => x.OrderIndex)
            .Select(section => new CourseSectionDto
            {
                Id = section.Id,
                CourseId = section.CourseId,
                Title = section.Title,
                OrderIndex = section.OrderIndex,

                Lessons = section.Lessons?
                    .OrderBy(x => x.OrderIndex)
                    .Select(lesson => new CourseLessonDto
                    {
                        Id = lesson.Id,
                        Title = lesson.Title,
                        Description = lesson.Description,
                        DurationSeconds = lesson.DurationSeconds,
                        IsFreePreview = lesson.IsFreePreview,
                        OrderIndex = lesson.OrderIndex,

                        EpisodeId = lesson.Episode == null
                            ? null
                            : lesson.Episode.Id,

                        VideoStatus = lesson.Episode == null
                            ? null
                            : lesson.Episode.Status.ToString(),

                        HasVideo = lesson.Episode != null
                    })
                    .ToList()
                    ?? new List<CourseLessonDto>()
            })
            .ToList()
            ?? new List<CourseSectionDto>();


        return new CourseDto
        {
            Id = c.Id,
            Title = c.Title,
            Description = c.Description,
            CoverImageUrl = c.CoverImageUrl,
            Price = c.Price,
            IsFree = c.IsFree,
            IsPublished = c.IsPublished,
            Level = c.Level,
            Category = c.Category,
            DurationMinutes = c.DurationMinutes,
            Sections = sections
        };
    }
}

using Handicraft.Application.Contracts.Persistence;
using Handicraft.Application.Contracts.Services;
using Handicraft.Application.Services;
using Handicraft.Domain.Entities;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

namespace Handicraft.Infrastructure.BackgroundServices;

public class VideoProcessingBackgroundService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<VideoProcessingBackgroundService> _logger;

    public VideoProcessingBackgroundService(
        IServiceScopeFactory scopeFactory,
        ILogger<VideoProcessingBackgroundService> logger)
    {
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = _scopeFactory.CreateScope();
                var episodeRepo = scope.ServiceProvider
                    .GetRequiredService<ICourseEpisodeRepository>();
                var storage = scope.ServiceProvider
                    .GetRequiredService<IVideoStorageService>();
                var processor = scope.ServiceProvider
                    .GetRequiredService<IVideoProcessor>();

                var pendingEpisodes = await episodeRepo.GetAsync(
                    e => e.Status == VideoProcessingStatus.Pending);

                foreach (var episode in pendingEpisodes)
                {
                    episode.Status = VideoProcessingStatus.Processing;
                    await episodeRepo.UpdateAsync(episode);

                    try
                    {
                        var rawPath = storage.GetRawVideoPath(
                           episode.LessonId, episode.Id, episode.RawVideoFileName!);

                        var hlsOutputDir = storage.GetHlsOutputDirectory(
                            episode.LessonId, episode.Id);

                        var (masterPlaylistRelativePath, durationSeconds) =
                            await processor.ConvertToHlsAsync(rawPath, hlsOutputDir, stoppingToken);

                        episode.VideoUrl = masterPlaylistRelativePath;
                        episode.DurationSeconds = durationSeconds;
                        episode.Status = VideoProcessingStatus.Ready;
                    }
                    catch (OperationCanceledException)
                    {
                        _logger.LogInformation(
                            "Video processing cancelled.");

                        break;
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(
                            ex,
                            "Failed to process episode {EpisodeId}",
                            episode.Id);

                        episode.Status = VideoProcessingStatus.Failed;
                    }

                    await episodeRepo.UpdateAsync(episode);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in VideoProcessingBackgroundService loop");
            }

            try
            {
                await Task.Delay(
                    TimeSpan.FromSeconds(10),
                    stoppingToken);
            }
            catch (TaskCanceledException)
            {
                _logger.LogInformation(
                    "Video processing service is stopping.");
            }
        }
    }
}

namespace Handicraft.Infrastructure.Repositories;

public class GenericRepository<T> : IGenericRepository<T> where T : class
{
    protected readonly AppDbContext _context;

    public GenericRepository(AppDbContext context)
    {
        _context = context;
    }

    public async Task<T?> GetByIdAsync(Guid id)
    {
        return await _context.Set<T>().FindAsync(id);
    }

    public async Task<IReadOnlyList<T>> GetAllAsync()
    {
        return await _context.Set<T>().ToListAsync();
    }

    public async Task<IReadOnlyList<T>> GetAsync(Expression<Func<T, bool>> predicate)
    {
        return await _context.Set<T>().Where(predicate).ToListAsync();
    }

    public async Task<T> AddAsync(T entity)
    {
        await _context.Set<T>().AddAsync(entity);
        await _context.SaveChangesAsync();
        return entity;
    }

    public async Task UpdateAsync(T entity)
    {
        _context.Entry(entity).State = EntityState.Modified;
        await _context.SaveChangesAsync();
    }

    public async Task DeleteAsync(T entity)
    {
        _context.Set<T>().Remove(entity);
        await _context.SaveChangesAsync();
    }

    public async Task<bool> ExistsAsync(Guid id)
    {
        var entity = await GetByIdAsync(id);
        return entity != null;
    }
}

using System.Diagnostics;
using Handicraft.Application.Contracts.Services;

namespace Handicraft.Infrastructure.Services;

public class FfmpegVideoProcessor : IVideoProcessor
{
    public async Task<VideoConversionResult> ConvertToHlsAsync(string inputFilePath, string outputDirectory, CancellationToken cancellationToken = default)
    {
        Directory.CreateDirectory(outputDirectory);

        var masterPlaylistPath = Path.Combine(outputDirectory, "master.m3u8");

        var ffmpegArgs =
            $"-i \"{inputFilePath}\" " +
            "-filter_complex \"[0:v]split=3[v1][v2][v3];" +
            "[v1]scale=w=640:h=360[v1out];[v2]scale=w=854:h=480[v2out];[v3]scale=w=1280:h=720[v3out]\" " +
            "-map [v1out] -c:v:0 libx264 -b:v:0 800k " +
            "-map [v2out] -c:v:1 libx264 -b:v:1 1400k " +
            "-map [v3out] -c:v:2 libx264 -b:v:2 2800k " +
            "-map a:0? -map a:0? -map a:0? -c:a aac -b:a 128k " +
            "-var_stream_map \"v:0,a:0 v:1,a:1 v:2,a:2\" " +
            "-f hls -hls_time 6 -hls_playlist_type vod " +
            $"-master_pl_name master.m3u8 " +
            $"-hls_segment_filename \"{outputDirectory}/v%v/segment%d.ts\" " +
            $"\"{outputDirectory}/v%v/playlist.m3u8\"";

        await RunProcessAsync("ffmpeg", ffmpegArgs, cancellationToken);

        var duration = await GetDurationSecondsAsync(inputFilePath, cancellationToken);

        return new VideoConversionResult("master.m3u8", duration);
    }

    private static async Task<int> GetDurationSecondsAsync(string inputFilePath, CancellationToken cancellationToken)
    {
        var args = $"-v error -show_entries format=duration -of default=noprint_wrappers=1:nokey=1 \"{inputFilePath}\"";
        var output = await RunProcessAsync("ffprobe", args, cancellationToken, captureOutput: true);

        return double.TryParse(output.Trim(), out var seconds) ? (int)seconds : 0;
    }

    private static async Task<string> RunProcessAsync(string fileName, string arguments, CancellationToken cancellationToken, bool captureOutput = false)
    {
        var psi = new ProcessStartInfo
        {
            FileName = fileName,
            Arguments = arguments,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true
        };

        using var process = Process.Start(psi)!;

        var stdOutTask = process.StandardOutput.ReadToEndAsync();
        var stdErrTask = process.StandardError.ReadToEndAsync();

        await process.WaitForExitAsync(cancellationToken);

        var stdOut = await stdOutTask;
        var stdErr = await stdErrTask;

        if (process.ExitCode != 0)
            throw new InvalidOperationException($"{fileName} failed: {stdErr}");

        return captureOutput ? stdOut : stdErr;
    }
}

using Handicraft.Application.Contracts.Services;
using Microsoft.Extensions.Configuration;

namespace Handicraft.Application.Services
{
    public class LocalVideoStorageService : IVideoStorageService
    {
        private readonly string _basePath;

        public LocalVideoStorageService(IConfiguration config)
        {
            _basePath = config["VideoStorage:BasePath"] ?? "/data/videos";
        }


        public async Task<string> SaveRawVideoAsync(
            Guid lessonId,
            Guid episodeId,
            Stream fileStream,
            string fileName,
            CancellationToken cancellationToken = default)
        {
            var folder = Path.Combine(
                _basePath,
                lessonId.ToString(),
                episodeId.ToString(),
                "raw");

            Directory.CreateDirectory(folder);

            var rawPath = Path.Combine(folder, fileName);

            await using var fs = new FileStream(
                rawPath,
                FileMode.Create);

            await fileStream.CopyToAsync(
                fs,
                cancellationToken);

            return rawPath;
        }


        public string GetRawVideoPath(
            Guid lessonId,
            Guid episodeId,
            string fileName)
        {
            return Path.Combine(
                _basePath,
                lessonId.ToString(),
                episodeId.ToString(),
                "raw",
                fileName);
        }


        public string GetHlsOutputDirectory(
            Guid lessonId,
            Guid episodeId)
        {
            var folder = Path.Combine(
                _basePath,
                lessonId.ToString(),
                episodeId.ToString(),
                "hls");

            Directory.CreateDirectory(folder);

            return folder;
        }


        public string GetHlsPhysicalFilePath(
            Guid lessonId,
            Guid episodeId,
            string relativePath)
        {
            return Path.Combine(
                GetHlsOutputDirectory(lessonId, episodeId),
                relativePath);
        }


        public string GetHlsPlaylistPath(
            Guid lessonId,
            Guid episodeId)
        {
            return Path.Combine(
                GetHlsOutputDirectory(lessonId, episodeId),
                "master.m3u8");
        }


        public Task DeleteEpisodeFilesAsync(
            Guid lessonId,
            Guid episodeId)
        {
            var folder = Path.Combine(
                _basePath,
                lessonId.ToString(),
                episodeId.ToString());

            if (Directory.Exists(folder))
                Directory.Delete(folder, recursive: true);

            return Task.CompletedTask;
        }
    }
}
